```bash
# Write .claude/rwenv-env file with resolved runtime config
# This is the single source of truth for hooks and agents at runtime
write_rwenv_env() {
    local dir="${1:-$PWD}"
    local rwenv_name="$2"
    local env_file="$dir/.claude/rwenv-env"
    local gitignore="$dir/.gitignore"

    # Load rwenv config
    local rwenv_config
    rwenv_config=$(get_rwenv_by_name "$rwenv_name") || {
        echo "ERROR: rwenv '$rwenv_name' not found" >&2
        return 1
    }

    # Determine mode
    local use_dev_container mode
    use_dev_container=$(get_use_dev_container)
    if [[ "$use_dev_container" == "true" ]]; then
        mode="container"
    else
        mode="local"
    fi

    # Read kubeconfig/context from the mode-specific sub-object
    local kubeconfig context
    kubeconfig=$(echo "$rwenv_config" | jq -r --arg mode "$mode" '.[$mode].kubeconfigPath // empty')
    context=$(echo "$rwenv_config" | jq -r --arg mode "$mode" '.[$mode].kubernetesContext // empty')

    if [[ -z "$kubeconfig" || -z "$context" ]]; then
        echo "ERROR: '$mode' sub-object missing or incomplete for rwenv '$rwenv_name'. Update envs.json." >&2
        return 1
    fi

    # Extract other fields
    local rwenv_type read_only gcp_project dev_container flux_repo_path
    rwenv_type=$(echo "$rwenv_config" | jq -r '.type')
    read_only=$(echo "$rwenv_config" | jq -r '.readOnly')
    gcp_project=$(echo "$rwenv_config" | jq -r '.gcpProject // empty')
    dev_container=$(get_dev_container)
    flux_repo_path="$(get_config_dir)/flux-repos/$rwenv_name"

    # Write env file
    mkdir -p "$dir/.claude"
    cat > "$env_file" <<ENVEOF
# Generated by /rwenv-set and /rwenv-local-mode. Do not edit manually.
RWENV_NAME=$rwenv_name
RWENV_TYPE=$rwenv_type
RWENV_MODE=$mode
RWENV_KUBECONFIG=$kubeconfig
RWENV_CONTEXT=$context
RWENV_READ_ONLY=$read_only
RWENV_GCP_PROJECT=$gcp_project
RWENV_DEV_CONTAINER=$dev_container
RWENV_FLUX_REPO=$flux_repo_path
ENVEOF

    # Auto-gitignore
    if [[ -d "$dir/.git" ]] || git -C "$dir" rev-parse --git-dir >/dev/null 2>&1; then
        if ! grep -qxF '.claude/rwenv-env' "$gitignore" 2>/dev/null; then
            if [[ -f "$gitignore" ]] && [[ -s "$gitignore" ]] && [[ "$(tail -c 1 "$gitignore" | wc -l)" -eq 0 ]]; then
                echo >> "$gitignore"
            fi
            echo '.claude/rwenv-env' >> "$gitignore"
        fi
    fi
}
```

### Task 4: Rewrite `transform-commands.sh` as validation-only hook

**Files:**
- Modify: `rwenv/hooks/transform-commands.sh`

**Step 1: Rewrite the hook**

Replace the full content. The new hook:
1. Sources `.claude/rwenv-env` from the working directory (`$PWD/.claude/rwenv-env`)
2. If env file missing, check if command is an rwenv command — if so block with error, otherwise pass through
3. Safety checks (same logic as before): read-only write detection, gcloud-for-k3s, gcloud writes always blocked
4. Dangerous docker ops: block stop/rm/kill/restart of `$RWENV_DEV_CONTAINER`
5. Validation — checks the command has correct flags/wrapping:
   - Container mode: any `kubectl|helm|flux|gcloud` in the command must appear after `docker exec.*$RWENV_DEV_CONTAINER` and must include `--kubeconfig` and `--context` (except gcloud which uses `--project`)
   - Local mode: any `kubectl|helm|flux|gcloud` must include `--kubeconfig` and `--context` (except gcloud which uses `--project`)
   - If validation fails: block with descriptive error
6. Auto-approve: output JSON with `permissionDecision: "allow"`

```bash
#!/usr/bin/env bash
# transform-commands.sh - Validation and auto-approval hook for rwenv
#
# This hook validates that kubectl/helm/flux/gcloud commands have correct
# flags and wrapping for the current mode, enforces safety, and auto-approves.
# It does NOT transform commands — agents construct mode-aware commands.

set -euo pipefail

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
PLUGIN_DIR="$(dirname "$SCRIPT_DIR")"

# Source utilities (for write-detection functions)
source "$PLUGIN_DIR/lib/rwenv-utils.sh"

# Commands that trigger rwenv validation
RWENV_COMMANDS="kubectl|helm|flux|gcloud|vault"

# Read JSON input from stdin
INPUT_JSON=$(cat)

# Extract the command
ORIGINAL_CMD=$(echo "$INPUT_JSON" | jq -r '.tool_input.command // empty')
if [[ -z "$ORIGINAL_CMD" ]]; then
    exit 0
fi

# Source the runtime env file
RWENV_ENV_FILE="${PWD}/.claude/rwenv-env"

# Check if command contains rwenv-managed commands
CONTAINS_RWENV_CMD=false
if echo "$ORIGINAL_CMD" | grep -qE "(^|[^a-zA-Z_-])($RWENV_COMMANDS)(\s|$)"; then
    CONTAINS_RWENV_CMD=true
fi

# If no env file, only block rwenv commands
if [[ ! -f "$RWENV_ENV_FILE" ]]; then
    if [[ "$CONTAINS_RWENV_CMD" == "true" ]]; then
        cat >&2 <<'EOF'
ERROR: No rwenv configured for this project.

Run: /rwenv-set <environment>
EOF
        exit 2
    fi
    exit 0
fi

# Source env vars
source "$RWENV_ENV_FILE"

# If command doesn't contain rwenv commands, check docker safety only
if [[ "$CONTAINS_RWENV_CMD" == "false" ]]; then
    # Check dangerous docker ops against dev container
    if [[ "$RWENV_MODE" == "container" ]] && [[ -n "$RWENV_DEV_CONTAINER" ]]; then
        DANGEROUS_DOCKER_OPS="stop|rm|remove|kill|restart|pause|update"
        if echo "$ORIGINAL_CMD" | grep -qE "docker ($DANGEROUS_DOCKER_OPS).*$RWENV_DEV_CONTAINER"; then
            cat >&2 <<EOF
ERROR: Cannot modify the dev container '$RWENV_DEV_CONTAINER'.
Blocked command: $ORIGINAL_CMD
EOF
            exit 2
        fi
        if echo "$ORIGINAL_CMD" | grep -qE "docker (stop|rm|kill|restart|pause) -a|docker container prune|docker system prune"; then
            cat >&2 <<EOF
ERROR: Cannot run bulk container operations.
Blocked command: $ORIGINAL_CMD
This could affect the dev container '$RWENV_DEV_CONTAINER'.
EOF
            exit 2
        fi
        # Auto-approve docker exec to dev container
        if echo "$ORIGINAL_CMD" | grep -qE "docker exec.*$RWENV_DEV_CONTAINER"; then
            echo "$INPUT_JSON" | jq '.hookSpecificOutput = {permissionDecision: "allow"}'
            exit 0
        fi
    fi
    exit 0
fi

# --- Safety checks ---

# gcloud for k3s: block
if echo "$ORIGINAL_CMD" | grep -qE "(^|[^a-zA-Z_-])gcloud(\s)" && [[ "$RWENV_TYPE" == "k3s" ]]; then
    cat >&2 <<EOF
ERROR: gcloud not available for k3s rwenv '$RWENV_NAME'.
Use a GKE rwenv for gcloud operations.
EOF
    exit 2
fi

# gcloud writes: always block
if echo "$ORIGINAL_CMD" | grep -qE "(^|[^a-zA-Z_-])gcloud(\s)"; then
    # Extract gcloud args (everything after gcloud)
    local_gcloud_args=$(echo "$ORIGINAL_CMD" | grep -oE "gcloud\s+.*" | head -1 | cut -d' ' -f2-)
    if is_gcloud_write_operation "$local_gcloud_args"; then
        cat >&2 <<EOF
ERROR: gcloud write operations are blocked for safety.
Blocked command: $ORIGINAL_CMD
EOF
        exit 2
    fi
fi

# Read-only enforcement for kubectl/helm/flux
if [[ "$RWENV_READ_ONLY" == "true" ]]; then
    # Check kubectl write ops
    if echo "$ORIGINAL_CMD" | grep -qE "(^|[^a-zA-Z_-])kubectl(\s)"; then
        local_kubectl_args=$(echo "$ORIGINAL_CMD" | grep -oE "kubectl\s+[^|;&]*" | head -1 | sed 's/kubectl\s*//')
        # Strip --kubeconfig and --context flags to get the actual subcommand
        local_kubectl_subcmd=$(echo "$local_kubectl_args" | sed 's/--kubeconfig=[^ ]* *//g; s/--context=[^ ]* *//g')
        if is_kubectl_write_operation "$local_kubectl_subcmd"; then
            cat >&2 <<EOF
ERROR: rwenv '$RWENV_NAME' is read-only. Cannot execute write operation.
Blocked command: $ORIGINAL_CMD
EOF
            exit 2
        fi
    fi

    # Check helm write ops
    if echo "$ORIGINAL_CMD" | grep -qE "(^|[^a-zA-Z_-])helm(\s)"; then
        local_helm_args=$(echo "$ORIGINAL_CMD" | grep -oE "helm\s+[^|;&]*" | head -1 | sed 's/helm\s*//')
        local_helm_subcmd=$(echo "$local_helm_args" | sed 's/--kubeconfig=[^ ]* *//g; s/--kube-context=[^ ]* *//g')
        if is_helm_write_operation "$local_helm_subcmd"; then
            cat >&2 <<EOF
ERROR: rwenv '$RWENV_NAME' is read-only. Cannot execute write operation.
Blocked command: $ORIGINAL_CMD
EOF
            exit 2
        fi
    fi

    # Check flux write ops
    if echo "$ORIGINAL_CMD" | grep -qE "(^|[^a-zA-Z_-])flux(\s)"; then
        local_flux_args=$(echo "$ORIGINAL_CMD" | grep -oE "flux\s+[^|;&]*" | head -1 | sed 's/flux\s*//')
        local_flux_subcmd=$(echo "$local_flux_args" | sed 's/--kubeconfig=[^ ]* *//g; s/--context=[^ ]* *//g')
        if is_flux_write_operation "$local_flux_subcmd"; then
            cat >&2 <<EOF
ERROR: rwenv '$RWENV_NAME' is read-only. Cannot execute write operation.
Blocked command: $ORIGINAL_CMD
EOF
            exit 2
        fi
    fi
fi

# --- Validation ---

# Validate kubectl/helm/flux have required flags
validate_has_flags() {
    local cmd_name="$1"
    local flag_pattern="$2"
    if echo "$ORIGINAL_CMD" | grep -qE "(^|[^a-zA-Z_-])${cmd_name}(\s)" && \
       ! echo "$ORIGINAL_CMD" | grep -qE "${cmd_name}.*${flag_pattern}"; then
        cat >&2 <<EOF
ERROR: $cmd_name command missing required flags.

Expected: --kubeconfig=\$RWENV_KUBECONFIG --context=\$RWENV_CONTEXT
Current mode: $RWENV_MODE

Source .claude/rwenv-env to get the correct values:
  RWENV_KUBECONFIG=$RWENV_KUBECONFIG
  RWENV_CONTEXT=$RWENV_CONTEXT
EOF
        exit 2
    fi
}

# Container mode: validate docker exec wrapping
if [[ "$RWENV_MODE" == "container" ]]; then
    for cmd in kubectl helm flux; do
        if echo "$ORIGINAL_CMD" | grep -qE "(^|[^a-zA-Z_-])${cmd}(\s)"; then
            # Must be inside docker exec to dev container
            if ! echo "$ORIGINAL_CMD" | grep -qE "docker exec.*$RWENV_DEV_CONTAINER.*${cmd}"; then
                cat >&2 <<EOF
ERROR: In container mode, $cmd must run inside docker exec $RWENV_DEV_CONTAINER.

Expected pattern:
  docker exec -i $RWENV_DEV_CONTAINER $cmd --kubeconfig=$RWENV_KUBECONFIG --context=$RWENV_CONTEXT ...
EOF
                exit 2
            fi
        fi
    done
    if echo "$ORIGINAL_CMD" | grep -qE "(^|[^a-zA-Z_-])gcloud(\s)"; then
        if ! echo "$ORIGINAL_CMD" | grep -qE "docker exec.*$RWENV_DEV_CONTAINER.*gcloud"; then
            cat >&2 <<EOF
ERROR: In container mode, gcloud must run inside docker exec $RWENV_DEV_CONTAINER.

Expected pattern:
  docker exec -i $RWENV_DEV_CONTAINER gcloud --project=$RWENV_GCP_PROJECT ...
EOF
            exit 2
        fi
    fi
fi

# Both modes: validate flags present
validate_has_flags "kubectl" "--kubeconfig=.*--context="
validate_has_flags "helm" "--kubeconfig=.*--kube-context="
validate_has_flags "flux" "--kubeconfig=.*--context="
if echo "$ORIGINAL_CMD" | grep -qE "(^|[^a-zA-Z_-])gcloud(\s)"; then
    if ! echo "$ORIGINAL_CMD" | grep -qE "gcloud.*--project="; then
        cat >&2 <<EOF
ERROR: gcloud command missing --project flag.

Expected: --project=$RWENV_GCP_PROJECT
EOF
        exit 2
    fi
fi

# --- Auto-approve ---
echo "$INPUT_JSON" | jq '.hookSpecificOutput = {permissionDecision: "allow"}'
```
